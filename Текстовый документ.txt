// --- Создание карточки ---
function createCard(title, content, img, fullHtml) {
  const previewText = content.replace(/<[^>]+>/g, "").slice(0, 90) + "...";
  const card = document.createElement("div");
  card.className = "news-card";
  card.innerHTML = `
    <img src="${img}" alt="${title}" />
    <div class="news-card-content">
      <h3>${title}</h3>
      <p>${previewText}</p>
    </div>
  `;
  card.addEventListener("click", () => openModal(fullHtml));
  return card;
}

function openModal(content) {
  modalBody.innerHTML = content;
  modal.style.display = "flex"; // показываем модалку
  stopAutoScroll();             // останавливаем автопрокрутку
}

function closeModal() {
  modal.style.display = "none";
  startAutoScroll();            // снова запускаем автопрокрутку
}

function loadNewsContent(lang) {
  let newsSlug;
  if (lang === "kk") newsSlug = "news-international-kaz";
  else if (lang === "ru") newsSlug = "news-international-rus";
  else if (lang === "en") newsSlug = "news-international-eng";
  else newsSlug = "news-international-rus";

  const carousel = document.getElementById("news-carousel"); // <-- обязательно нужно определить carousel

  if (!carousel) {
    console.error("Элемент #carousel не найден");
    return;
  }

  // Очищаем старый контент перед загрузкой
  carousel.innerHTML = "";

  fetch(`/api/get-page/?slug=${newsSlug}`)
    .then(res => {
      if (!res.ok) throw new Error("Page not found");
      return res.json();
    })
    .then(data => {
      const contentHtml = data.content || "";
      const parser = new DOMParser();
      const regex = /<h1>(.*?)<\/h1>([\s\S]*?)(?=<h1>|$)/gi;
      let match;

      while ((match = regex.exec(contentHtml)) !== null) {
        const title = match[1].trim();
        const bodyHtml = match[2].trim();
        const bodyDoc = parser.parseFromString(bodyHtml, "text/html");
        const img = bodyDoc.querySelector("img")?.src || "https://via.placeholder.com/220";
        const text = bodyDoc.querySelector("p")?.innerHTML || "";

        // Предполагается, что у вас есть функция createCard(title, text, img, fullHtml)
        carousel.appendChild(createCard(title, text, img, `<h1>${title}</h1>${bodyHtml}`));
      }

      // --- Запуск автопрокрутки с задержкой ---
      setTimeout(startAutoScroll, 100);
    })
    .catch(err => {
      console.error(err);
      carousel.innerText = err.message || err;
    });
}

  // обработка кликов по кнопкам языков
document.querySelectorAll(".lang-btn").forEach(btn => {
  btn.addEventListener("click", () => {
    // убираем активный класс со всех
    document.querySelectorAll(".lang-btn").forEach(b => b.classList.remove("active"));

    // добавляем активный класс выбранной кнопке
    btn.classList.add("active");

    // сохраняем текущий язык
    let current = btn.getAttribute("data-lang");

    // загружаем контент новостей на этом языке
    loadNewsContent(current);
  });
});

// --- Автопрокрутка ---
function startAutoScroll() {
  if (!carousel || carousel.children.length === 0) return;

  // клонируем все карточки для бесконечной прокрутки
  const items = Array.from(carousel.children);
  items.forEach(item => carousel.appendChild(item.cloneNode(true)));

  let translateY = 0;

  function scroll() {
    translateY += scrollSpeed;

    // когда прокрутили половину (оригинальные карточки), сбрасываем
    if (translateY >= carousel.scrollHeight / 2) translateY = 0;

    carousel.style.transform = `translateY(-${translateY}px)`;
    requestAnimationFrame(scroll);
  }

  requestAnimationFrame(scroll);
}
